1. Priprema environmenta

cd ClusterSizerv2
python -m venv build-venv
build-venv\Scripts\activate

pip install -r requirements.txt
pip uninstall PySide6 PySide6-Addons -y
pip install PySide6-Essentials
pip install pyinstaller

2 Build

pyinstaller --name ClusterSizerv2 --onedir --windowed --noconfirm --clean `
  --exclude-module tkinter --exclude-module unittest --exclude-module test `
  --exclude-module pydoc --exclude-module doctest `
  main.py


3. CUT

Remove-Item "dist\ClusterSizer\_internal\opengl32sw.dll" -ErrorAction SilentlyContinue

$plugins = "dist\ClusterSizer\_internal\PySide6\plugins"
Remove-Item "$plugins\tls" -Recurse -ErrorAction SilentlyContinue
Remove-Item "$plugins\networkinformation" -Recurse -ErrorAction SilentlyContinue
Remove-Item "$plugins\generic" -Recurse -ErrorAction SilentlyContinue
Remove-Item "$plugins\iconengines" -Recurse -ErrorAction SilentlyContinue

4. Github release
Compress-Archive -Path "dist\ClusterSizer\*" -DestinationPath "ClusterSizer-VERSION-win64.zip"

5. Github
GitHub repo → Releases → Draft a new release → tag v2.4.2 → upload .zip as asset → Publish.
